#!/usr/bin/env python3 """ 公式サイト（聖徳自動車学園）の料金ページを取得し、 表()の中身を price-data.json に書き出すスクリプト。 GitHub Actions から定期的に実行される想定。 公式サイトの構造が変わって表がうまく取れなかった場合は、 既存の price-data.json を壊さないよう、何もせずエラー終了する。 """ import json import re import sys from datetime import datetime, timezone from pathlib import Path import requests from bs4 import BeautifulSoup SOURCE_URL = "https://www.shotoku-ds.net/guidance/price.html" OUTPUT_PATH = Path(__file__).resolve().parent.parent / "price-data.json" # 全角・半角どちらの数字にもマッチ（価格らしい表かどうかの簡易判定に使う） DIGIT_RE = re.compile(r"[0-9０-９]") def looks_like_price_table(rows: list[list[str]]) -> bool: """数字が1つでも含まれていれば「価格っぽい表」とみなす（記号は問わない）""" for row in rows: for cell in row: if DIGIT_RE.search(cell): return True return False def parse_table_grid(table_el): """を出現順に走査しつつ rowspan/colspan を展開し、 すべての行が同じ列数になるような行列（グリッド）を作る。 公式サイトの料金表は「車種」「所持免許」などのセルが rowspan/colspanで複数行にまたがっているため、単純に ごとのセルを並べるだけだと、またがっている行のセルが 抜け落ちて列がズレてしまう（例：夜間の行だけ車種・所持免許が 無くなり、他の列の値になってしまう）。 このため、ブラウザが行う表描画と同じロジック （rowspan/colspanを考慮したセル展開）をここでも行う。 """ trs = table_el.find_all("tr") grid = [] pending = {} # 列インデックス -> [残り行数, 値]（rowspanで持ち越し中のセル） for tr in trs: cells = tr.find_all(["td", "th"]) row_out = [] col = 0 cell_idx = 0 while cell_idx < len(cells) or col in pending: if col in pending: remaining, value = pending[col] row_out.append(value) if remaining <= 1: del pending[col] else: pending[col] = [remaining - 1, value] col += 1 continue cell = cells[cell_idx] text = cell.get_text(strip=True) try: colspan = int(cell.get("colspan", 1)) except (TypeError, ValueError): colspan = 1 try: rowspan = int(cell.get("rowspan", 1)) except (TypeError, ValueError): rowspan = 1 colspan = max(colspan, 1) rowspan = max(rowspan, 1) for i in range(colspan): row_out.append(text) if rowspan > 1: pending[col + i] = [rowspan - 1, text] col += colspan cell_idx += 1 grid.append(row_out) # 念のため、行ごとの列数を最大値にそろえる（本来は揃っているはず） max_cols = max((len(r) for r in grid), default=0) for r in grid: while len(r) < max_cols: r.append("") return grid, trs def extract_tables(html_bytes: bytes): # resp.text ではなく生バイト列を渡し、BeautifulSoup自身に文字コードを # 判定させる（requestsのヘッダーベースの文字コード判定は誤検出しやすいため）。 soup = BeautifulSoup(html_bytes, "lxml") all_tables = soup.find_all("table") print(f"[INFO] ページ内の
要素数: {len(all_tables)}", file=sys.stderr) # ページ内の見出し(h1-h6)とtableを出現順にまとめて走査し、 # 各tableの直前に出てきた見出しをキャプションとして扱う。 heading_tags = {"h1", "h2", "h3", "h4", "h5", "h6"} current_caption = "" results = [] skipped = 0 for el in soup.find_all(list(heading_tags) + ["table"]): if el.name in heading_tags: text = el.get_text(strip=True) if text: current_caption = text continue # el.name == "table" grid, trs = parse_table_grid(el) headers = [] table_rows = grid if trs: first_tr_cells = trs[0].find_all(["td", "th"]) has_header = bool(first_tr_cells) and all( c.name == "th" for c in first_tr_cells ) if has_header and grid: headers = grid[0] table_rows = grid[1:] # 完全に空の行は除外 table_rows = [row for row in table_rows if any(cell.strip() for cell in row)] preview = table_rows[0] if table_rows else [] print( f"[INFO] table candidate: caption='{current_caption}' " f"headers={headers} rows={len(table_rows)} first_row={preview}", file=sys.stderr, ) if not table_rows: skipped += 1 continue if not looks_like_price_table(table_rows): skipped += 1 continue results.append({ "caption": current_caption or "料金表", "headers": headers, "rows": table_rows, }) print( f"[INFO] 採用: {len(results)} 個 / 除外: {skipped} 個", file=sys.stderr, ) return results def main(): try: resp = requests.get( SOURCE_URL, timeout=20, headers={"User-Agent": "Mozilla/5.0 (compatible; ShotokuPriceSync/1.0)"}, ) resp.raise_for_status() except Exception as e: print(f"[ERROR] 公式サイトの取得に失敗しました: {e}", file=sys.stderr) sys.exit(1) tables = extract_tables(resp.content) # 安全チェック：1個も取れていない場合のみ中止（サイトの完全な構造変化などを想定）。 MIN_TABLES = 1 if len(tables) < MIN_TABLES: print( f"[ERROR] 料金表が1つも取得できませんでした。" "公式サイトの構造が変わった可能性があるため、更新を中止します。", file=sys.stderr, ) sys.exit(1) data = { "updated_at": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"), "source_url": SOURCE_URL, "tables": tables, } OUTPUT_PATH.write_text( json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8" ) print(f"[OK] {len(tables)} 個の料金表を price-data.json に書き出しました。") if __name__ == "__main__": main()
